% _tmp.tex - 单章独立编译模板（preamble 唯一真源）
%
% 本文件是单章编译 preamble 的唯一真源。fix.py 读取本文件，将占位符
% CHAPTER_TEX 替换为实际章节中间文件名（chN.tex / 附录X_ch.tex），
% 写出章节目录下的 _tmp.tex 并交给 xelatex 编译。作者一般不需要手动改动本文件。
%
% 它与整书模板 main.tex 同步关键补丁：中文 URL 支持、参考文献降级、
% 代码块环境、图片宽度上限、防溢出。集中放这里，章节源文件保持干净。
% 注意：不要在 fix.py 内再内联一份副本，否则会双源漂移。

\documentclass[a4paper, 11pt, twoside]{ctexrep}
\usepackage{amsmath,amssymb}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{multirow}
\usepackage{array}
\usepackage{longtable}
\usepackage{tabularx}
\usepackage[unicode]{hyperref}
\usepackage{indentfirst}
\usepackage{xcolor}
\usepackage{xeCJK}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning, fit, backgrounds, calc}
\usepackage{adjustbox}
\usepackage{tcolorbox}
\tcbuselibrary{listings, skins, breakable}
\usepackage{enumitem}
\usepackage{url}
\usepackage{etoolbox}
\usepackage{geometry}
\geometry{margin=2.5cm}
\usepackage{fancyhdr}

% === 页面版式：与整书 main.tex 保持一致（twoside + 2.5cm + fancyhdr） ===
% 否则单章 PDF 与主控文件格式不一致（用户反馈的合并质量问题之一）。
\setlength{\headheight}{15pt}
\pagestyle{fancy}
\fancyhead[LE,RO]{\slshape\nouppercase{\leftmark}}
\fancyhead[LO,RE]{\slshape\nouppercase{\rightmark}}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

% === 共用补丁片段（唯一真源：templates/_preamble_shared.tex） ===
% fix 会把本片段内联进模板的对应占位处（中文 URL 支持 / 参考文献降级 /
% codeblock 环境 / 防溢出），避免与 main.tex 双份手维护导致漂移。
{{SHARED_PREAMBLE}}

\XeTeXlinebreaklocale "zh"
\XeTeXlinebreakskip=0pt
\graphicspath{{figures/}}
\DeclareGraphicsExtensions{.pdf,.png,.jpg}
\hypersetup{colorlinks=true, linkcolor=blue, urlcolor=blue, citecolor=blue}

\begin{document}
\input{CHAPTER_TEX}    % ← fix.py 会把这一行替换为实际章节中间文件（ch1.tex / 附录A_ch.tex 等）
\end{document}
